 71    def estimate_speed_profile(self) -> list:
 72        """
 73        Estimate speed profile
 74
 75        :return:
 76        """
 77        # Iterate over the segments
 78        for i in range(0, len(self._route.segment_start_point) - 1):
 79            # Coincides with the limit speed of the following section
 80            if i < self._num_segments:
 81                # Segment end point
 82                segment_end_point = self._route.segment_start_point[i + 1]
 83                # Final speed of the segment (m/s) is the next one
 84                final_speed = self._route.speed_limit_km_h[i + 1] / 3.6
 85
 86            # Last segment, arrival at destination
 87            else:
 88                # Segment end point
 89                segment_end_point = max(self._route.segment_start_point)
 90                # Final speed of the segment (m/s) is the actual one
 91                final_speed = self._route.speed_limit_km_h[i] / 3.6
 92
 93            # Segment length (m)
 94            segment_length = segment_end_point - self._route.segment_start_point[i]
 95            # Calculation of speed limit
 96            speed_limit = final_speed
 97
 98            self._delta_t.append(float(segment_length / speed_limit))
 99
100            # The kinematics of the vehicle is calculated at each simulation step.
101            while self._space[self._step - 1] <= segment_end_point:
102                self._slope_t.append(self._route.slope[i])
103                # Braking distance calculation
104                brake_distance = (final_speed ** 2 - self._speed_m_s[self._step - 1] ** 2) / (
105                        2 * self._vehicle.ax_brake)
106
107                # Determination of state (acceleration / constant speed / braking)
108                # If it is in acceleration, state=1, so a=ax_trac
109                # If it has reached the maximum speed of the section, state=0. In this case the speed is constant and
110                # a=0.
111                # If it must start braking, the state=-1, so a=ax_brake
112
113                # Condition of speed greater than the final speed. In this case it makes sense that braking may exist.
114                if self._speed_m_s[self._step - 1] >= final_speed:
115                    #  You do not have to brake, the current position indicates that you have not reached the braking
116                    #  starting point.
117                    if self._space[self._step - 1] < (segment_end_point - brake_distance):
118                        # Definition of state-dependent acceleration
119                        # If you do not reach the speed limit, you can speed up
120                        if self._speed_m_s[self._step - 1] < speed_limit:
121                            # Acceleration is smoothed, in case of being close to the speed limit.
122                            factor = calculate_acceleration_factor(self._speed_m_s[self._step - 1], speed_limit)
123                            # Calculate acceleration
124                            if self._speed_m_s[self._step - 1] < (self._vehicle.v1_km_h / 3.6):
125                                a = factor * self._ax_trac[1]
126                            elif self._speed_m_s[self._step - 1] > (self._vehicle.v2_km_h / 3.6):
127                                a = factor * self._ax_trac[3]
128                            else:
129                                a = factor * self._ax_trac[2]
130                            # State = 1 -> acceleration
131                            self._state.append(1)
132                        else:
133                            # If the speed limit of the section has been reached, the speed is kept constant,
134                            # This implies that the acceleration is zero.
135                            if self._speed_m_s[self._step - 1] > speed_limit:
136                                # In this case you have to brake up to the maximum speed of the section
137                                factor = calculate_acceleration_factor(self._speed_m_s[self._step - 1], speed_limit)
138                                # Calculate acceleration
139                                a = factor * self._vehicle.ax_brake
140                                # State = -1 -> braking
141                                self._state.append(-1)
142                            else:
143                                # Acceleration equal to 0
144                                a = 0
145                                # State = 0 -> Maximum speed reached
146                                self._state.append(0)
147                    # It is in braking
148                    else:
149                        # Acceleration is smoothed, in case of being close to the speed limit.
150                        factor = calculate_acceleration_factor(self._speed_m_s[self._step - 1], speed_limit)
151                        # Calculate acceleration
152                        a = factor * self._vehicle.ax_brake
153                        # State = -1 -> braking
154                        self._state.append(-1)
155                # In case of speed(self._step)<final_speed: No braking
156                else:
157                    # Definition of acceleration as a function of sub-segment
158                    if self._speed_m_s[self._step - 1] < speed_limit:
159                        # Acceleration is smoothed, in case of being close to the speed limit.
160                        factor = calculate_acceleration_factor(self._speed_m_s[self._step - 1], speed_limit)
161                        # Calculate acceleration
162                        if self._speed_m_s[self._step - 1] < (self._vehicle.v1_km_h / 3.6):
163                            a = factor * self._ax_trac[1]
164                        elif self._speed_m_s[self._step - 1] > (self._vehicle.v2_km_h / 3.6):
165                            a = factor * self._ax_trac[3]
166                        else:
167                            a = factor * self._ax_trac[2]
168                        # State = 1 -> acceleration
169                        self._state.append(1)
170                    # Speed limit case
171                    else:
172                        # Acceleration equal to 0
173                        a = 0
174                        # State = 0 -> Maximum speed reached
175                        self._state.append(0)
176
177                # Acceleration value is updated
178                if a > self._acceleration_limit:
179                    a = self._acceleration_limit
180
181                # A speed is calculated as a function of the acceleration imposed in the previous steps
182                speed_calc = max(self._speed_m_s[self._step - 1] + a * self._delta_t[i], 0)
183                # Speed is smoothed to avoid oscillations
184                if speed_calc - speed_limit < 0.1 or speed_calc > speed_limit:
185                    self._speed_m_s.append(speed_limit)
186                else:
187                    self._speed_m_s.append(speed_calc)
188
189                # Also the space is calculated as a function of the smoothed speed
190                self._space.append(self._space[self._step - 1] + self._speed_m_s[self._step] *
191                                   self._delta_t[i] + 0.5 * a * (self._delta_t[i] ** 2))
192
193                # Time is updated
194                self._time.append(self._time[self._step - 1] + self._delta_t[i])
195                # Step is updated
196                self._step += 1
197            # print(contador)
198
199        # The only relevant feature here is the speed
200        return self._speed_m_s

202    def calculate_acceleration(self):
203        self._acceleration = [0]  # Restart to 0
204        # Parse speed to km/h
205        speed_km_h = [speed * 3.6 for speed in self._speed_m_s]
206
207        # Iterate over all number of segments (e.g. retrieved from length of speed
208        for i in range(1, len(speed_km_h)):
209
210            if self._time[i] - self._time[i - 1] > 0:
211                # Calculate acceleration value with the difference of speeds in time
212                value = (1 / 3.6) * (speed_km_h[i] - speed_km_h[i - 1]) / (self._time[i] - self._time[i - 1])
213            else:
214                value = 0
215
216            if value > self._acceleration_limit:
217                self._acceleration.append(self._acceleration_limit)
218            else:
219                self._acceleration.append(value)
220
221        return self._acceleration

223    def calculate_resistances(self):
224        self._resistances = [0]  # Restart to 0
225        # Parse speed to km/h
226        speed_km_h = [speed * 3.6 for speed in self._speed_m_s]
227
228        # Iterate over all number of segments (e.g. retrieved from length of speed
229        for i in range(len(self._speed_m_s)):
230            self._resistances.append(self._vehicle.A + self._vehicle.B * speed_km_h[i] +
231                                     self._vehicle.C * (speed_km_h[i] * speed_km_h[i]))

235    def calculate_slopes_instant(self, heights):
236        """
237        Calculate slopes at each instant. Used only for experiments
238        :param heights:
239        :return:
240        """
241        self._slope_t = [0]  # Restart to 0
242        # Parse speed to km/h
243        speed_km_h = [speed * 3.6 for speed in self._speed_m_s]
244
245        for i in range(1, len(heights)):
246            segment_length = (0.5 / 3.6) * (speed_km_h[i] + speed_km_h[i - 1]) * \
247                             (self._time[i] - self._time[i - 1])
248            delta_h = heights[i] - heights[i - 1]
249
250            if segment_length == 0:
251                self._slope_t.append(0)
252            else:
253                slope_value = delta_h / segment_length
254                if abs(slope_value) > 0.2:
255                    if speed_km_h[i] < 5:
256                        self._slope_t.append(0)
257                    else:
258                        self._slope_t.append(0.2 * slope_value / abs(slope_value))
259                else:
260                    self._slope_t.append(slope_value)

262    def calculate_gravitational_resistances(self):
263        self._gravitational_resistances = [0]  # Restart to 0
264
265        for i in range(len(self._speed_m_s)):
266            self._gravitational_resistances.append(GRAVITY * self.vehicle.total_mass * self._slope_t[i])
267        return self._gravitational_resistances
